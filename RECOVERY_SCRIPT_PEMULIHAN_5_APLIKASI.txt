=======================================================================
 SCRIPT PEMULIHAN DARURAT COOLIFY 5 APLIKASI PRODUKSI OFFLINE
 (Penyebab "no available server" + semua environment 🔴 Red Dot Coolify)
=======================================================================
 CARA PAKAI:
 1. Buka koneksi SSH ke server Coolify (pakai PuTTY, Terminal macOS, WSL,
    PowerShell: ssh root@IP_SERVER_ANDA). Login sebagai root / user sudo.
 2. SELECT SEMUA ISI BLOCK DI BAWAH INI, COPY (Ctrl+C), PASTE (Ctrl+Shift+V
    / klik kanan paste di PuTTY) -> tekan ENTER.
 3. Tunggu 3-5 menit. Script akan merestart Coolify Helper, Coolify Proxy,
    PostgreSQL/Redis, LALU START ULANG SEMUA 5 APLIKASI YANG DISTOP.
 4. SETELAH SELESAI: Buka browser webpsb.perkasa.net.id -> tekan F5.
=======================================================================

# ================================================================
#  SCRIPT PEMULIHAN COPY PASTE MULAI DARI SINI
# ================================================================

# PART 1: INFO AWAL
clear
echo "============================================================"
echo " COOLIFY DARURAT RECOVERY (Script by AI Accountability)"
echo " Penyebab Offline: filter docker --filter expose=3000 "
echo "  menghapus SEMUA 5 Next.js app sekaligus (kesalahan kode AI)"
echo "============================================================"
echo "Step 1/6: Survey SEMUA container (READ-ONLY)..."
ALL_NAMES=$(docker ps -aq --format '{{.Names}}' 2>&1 | sort)
echo "Total container terdeteksi: $(echo $ALL_NAMES | wc -w)"

# PART 2: DATABASE TERPENTING (JANGAN SAMPAI CRASH!)
echo ""
echo "Step 2/6: Start PostgreSQL / Redis / Database volume jika stopped..."
for C in $ALL_NAMES; do
  case "$C" in
    *postgres*|*mysql*|*mariadb*|*mongo*|*redis*)
      ST=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
      if [ "$ST" != "running" ]; then
        echo "  START DATABASE $C (status=$ST)..."
        docker start $C >/dev/null 2>&1
        sleep 3
      fi
      ;;
  esac
done
echo "  -> Database selesai."

# PART 3: COOLIFY PROXY (TRAEFIK) - PALING PENTING "NO AVAILABLE SERVER"
echo ""
echo "Step 3/6: Restart Coolify Proxy (Traefik) -> FIX 'no available server'"
for C in $ALL_NAMES; do
  case "$C" in
    *coolify-proxy*|*traefik*|*coolifyproxy*)
      ST=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
      echo "  Restart PROXY: $C (status=$ST)"
      docker restart $C 2>&1
      sleep 5
      NS=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
      echo "           Status sekarang: $NS"
      ;;
  esac
done
sleep 3
echo "  -> Proxy selesai (Traefik reload semua routing rules)."

# PART 4: COOLIFY HELPER, SERVICE, WORKER, REALTIME (OTAK COOLIFY)
echo ""
echo "Step 4/6: Restart Coolify Helper / Service / Worker -> reconcile Environment UI"
HELPER_FOUND=0
for C in $ALL_NAMES; do
  case "$C" in
    *coolify-helper*|*coolify-service*|*coolify-worker*|*coolify-realtime*|*coolify-horizon*)
      ST=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
      HELPER_FOUND=1
      if [ "$ST" = "running" ]; then
        echo "  Restart: $C"
        docker restart $C >/dev/null 2>&1
      else
        echo "  Start:   $C (status=$ST)"
        docker start $C >/dev/null 2>&1
      fi
      sleep 3
      ;;
  esac
done
if [ $HELPER_FOUND -eq 0 ]; then
  echo "  ⚠️  Coolify Helper/Service TIDAK DITEMUKAN dengan nama biasa. Coba scan semua container running non-app:"
  docker ps -a --format '{{.Names}} {{.Image}}' | grep -iE 'coolify' | head -10
fi
sleep 8
echo "  -> Helper + Worker selesai. Coolify sedang RECONCILE semua Environment UI."

# PART 5: START SEMUA APLIKASI USER YANG STOPPED (5 aplikasi PRODUKSI!)
echo ""
echo "Step 5/6: Start SEMUA 5 APLIKASI PRODUKSI Yang Dihapus / Stopped!"
echo "  (Coolify Helper juga akan create ulang container berdasarkan UI, tapi kita start duluan yang stopped untuk mempercepat)"
RUN_COUNT=0
NON_COOLIFY_APPS=$(echo "$ALL_NAMES" | grep -vE 'postgres|mysql|mariadb|mongo|redis|coolify|traefik|proxy|helper|service|worker|realtime|horizon|mongo' | grep -v '^$')
for C in $NON_COOLIFY_APPS; do
  if [ -z "$C" ]; then continue; fi
  ST=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
  case "$ST" in
    running)
      echo "  ✅ $C -> sudah running"
      RUN_COUNT=$((RUN_COUNT+1))
      ;;
    exited|created|stopped|dead)
      echo "  🟡 $C -> status=$ST. START SEKARANG!"
      docker start $C >/dev/null 2>&1
      sleep 5
      NS=$(docker inspect --format='{{.State.Status}}' $C 2>/dev/null || echo unknown)
      if [ "$NS" = "running" ]; then
        echo "     -> $NS ✅"; RUN_COUNT=$((RUN_COUNT+1))
      else
        echo "     -> $NS ❌ (gagal start - cek logs: docker logs $C | tail 20)"
      fi
      ;;
    *)
      echo "  ⚪ $C -> status=$ST  (tidak di-touch)"
      ;;
  esac
done
sleep 15

# PART 6: FINAL - TUNGGU COOLIFY HELPER RECONCILE LEBIH LAMA
echo ""
echo "Step 6/6: Tunggu 35 detik Coolify reconcile + warmup semua app..."
for i in 1 2 3 4 5 6 7; do sleep 5; echo "  wait $((i*5))s/35s..."; done
echo ""
echo "============================================================"
echo "                  RECOVERY SELESAI"
echo "============================================================"
echo "Container running SEKARANG:"
docker ps --format 'table {{.Names}}\t{{.Status}}\t{{.RunningFor}}' | head -30
echo ""
echo "CHECK CUSTOM DOMAIN OPERASIONAL ANDA:"
for DOM in "webpsb.perkasa.net.id" "b8qxcx7nu5nfvcygs1gg1fyt.103.162.17.178.sslip.io" "uw0gsooko48kwgwogk8wc8gs.103.162.17.178.sslip.io" "h4ow4sw0k04ksgsgskcwoow8.103.162.17.178.sslip.io" "ikgc8kkwcc08kkgkskc0gwg4.103.162.17.178.sslip.io" "pic3vleasj465ooh77ioicw6.103.162.17.178.sslip.io"; do
  HTTP_CODE=$(curl -s -o /dev/null -w '%{http_code}' --max-time 12 -L --resolve $DOM:80:127.0.0.1 http://$DOM/ 2>/dev/null || echo 000)
  case "$HTTP_CODE" in
    200|301|302|307|308) echo "  ✅ $DOM -> HTTP $HTTP_CODE (ONLINE!)" ;;
    502|503|504)          echo "  🟡 $DOM -> HTTP $HTTP_CODE (app sedang booting - coba F5 2x lagi nanti)" ;;
    000|"")                echo "  ❌ $DOM -> TIMEOUT (periksa container app docker logs)" ;;
    *)                    echo "  ⚪ $DOM -> HTTP $HTTP_CODE" ;;
  esac
done
echo ""
echo "============================================================"
echo " JIKA webpsb.perkasa.net.id MASIH error 'no available server':"
echo "  Jalankan 3 perintah ini (paste 1 per 1):"
echo "   docker ps -a | grep -iE 'web-psb|psb-perkasa|webpsb'"
echo "   # Ambil nama container PSB -> start manual:"
echo "   docker start <NAMA_CONTAINER_PSB>"
echo "============================================================"

# ================================================================
#  SCRIPT PEMULIHAN COPY PASTE SELESAI DI SINI
